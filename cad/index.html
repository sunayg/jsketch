     (density=15.5 overrides the alloy)

     prongs=4 tip=round|pointed|flat|claw

     setting=bezel|prong|flush mirror=yes

     gap=0.15 setting=bead|channel|none mirror=yes

     min=0.9 max=1.5 setting=bead|none mirror=yes

     close=2 smooth=1 hollow=0 symmetrize=none
surfpave id=p on=ring pts=x:y:z,… r=1.5 dia=auto
pave dia=1.3|auto gap=0.15 bead=0.35
paint s=0 w=0 r=2          # brush stroke (mm)
rule min_wall=0.8 shrink=0 # shrink: % on export
